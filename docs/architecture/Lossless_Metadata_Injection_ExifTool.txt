import exiftool

def inject_metadata_lossless(file_path: str, title: str, description: str, keywords: list):
    """
    Injects metadata generated by Claude into the original photo 
    without altering a single pixel or losing visual quality.
    """
    # Convert the keyword list into a comma-separated string
    keywords_str = ", ".join(keywords)
    
    with exiftool.ExifToolHelper() as et:
        et.set_tags(
            [file_path],
            tags={
                "IPTC:ObjectName": title,            # Photo title
                "IPTC:Caption-Abstract": description, # Description
                "IPTC:Keywords": keywords,           # Keyword list
                "XMP:Description": description,       # XMP format backup
                "XMP:Subject": keywords
            },
            params=["-overwrite_original"] # Prevents generating duplicate .original files
        )
    print(f"✅ Lossless metadata successfully injected into: {file_path}")
