from datetime import datetime
from enum import Enum
from uuid import uuid4
from sqlalchemy import Column, String, DateTime, ForeignKey, Boolean, Text, Integer, Enum as SQLEnum
from sqlalchemy.orm import declarative_base, relationship

Base = declarative_base()

class UploadStatus(str, Enum):
    PENDING = "pending"       # Queued, awaiting processing
    PROCESSING = "processing" # AI generating and analyzing metadata
    READY = "ready"           # Metadata ready, awaiting FTP upload
    UPLOADING = "uploading"   # Actively transferring to stock agencies
    COMPLETED = "completed"   # Successful upload confirmed by server
    FAILED = "failed"         # Error encountered during processing or uploading


class User(Base):
    __tablename__ = "users"

    id = Column(String(36), primary_key=True, default=lambda: str(uuid4()))
    email = Column(String(255), unique=True, nullable=False, index=True)
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # Agency credentials ENCRYPTED with the AES-256 module
    adobe_ftp_user = Column(String(255), nullable=True)
    adobe_ftp_pass_encrypted = Column(Text, nullable=True)
    
    shutterstock_ftp_user = Column(String(255), nullable=True)
    shutterstock_ftp_pass_encrypted = Column(Text, nullable=True)

    # Relationships
    photos = relationship("Photo", back_populates="user", cascade="all, delete-orphan")


class Photo(Base):
    __tablename__ = "photos"

    id = Column(String(36), primary_key=True, default=lambda: str(uuid4()))
    user_id = Column(String(36), ForeignKey("users.id", ondelete="CASCADE"), nullable=False, index=True)
    
    # Cloud storage paths (e.g., AWS S3 / Google Cloud Storage)
    original_storage_url = Column(Text, nullable=False)
    thumbnail_storage_url = Column(Text, nullable=False) # Lightweight version processed by AI
    
    # SEO Metadata generated by Claude (populated during 'PROCESSING' status)
    title = Column(String(255), nullable=True)
    description = Column(Text, nullable=True)
    keywords = Column(Text, nullable=True) # Stored as comma-separated values or JSON string
    requires_model_release = Column(Boolean, default=False)
    
    # State control for process queueing
    status = Column(SQLEnum(UploadStatus), default=UploadStatus.PENDING, index=True)
    error_log = Column(Text, nullable=True) # Stores the traceback/logs if process fails
    
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

    # Relationships
    user = relationship("User", back_populates="photos")
